function doPost(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); // 1. 메인 시트 지정 (시트1) var mainSheet = ss.getSheetByName("시트1") || ss.getSheets()[0]; // 2. 안전화/유니폼 전용 시트 var subSheet = ss.getSheetByName("안전화유니폼"); if (!subSheet) { subSheet = ss.insertSheet("안전화유니폼"); subSheet.appendRow(["제출시간", "소속", "이름(한글)", "생년월일", "휴대폰", "안전화사이즈", "유니폼사이즈-상", "유니폼사이즈-하"]); } var data = JSON.parse(e.postData.contents); var submitTime = new Date(); // --- [파트 A] 기존에 삽입되어 있던 이미지 모두 삭제 (누적 방지) --- var images = mainSheet.getImages(); for (var i = 0; i < images.length; i++) { var imgRange = images[i].getAnchorCell(); var row = imgRange.getRow(); var col = imgRange.getColumn(); if (row >= 12 && row <= 13 && col >= 1 && col <= 8) { images[i].remove(); } } // --- [파트 B] 메인 시트에 데이터 입력 --- mainSheet.getRange("B4:E5").merge().setValue(data.department); mainSheet.getRange("G4:H4").setValue(data.hireDate); mainSheet.getRange("G5:H5").setValue(data.appointDate); mainSheet.getRange("B7:D7").merge().setValue(data.nameKo); mainSheet.getRange("E7:H7").merge().setValue(data.nameEn); mainSheet.getRange("B8:D9").merge().setValue(data.phone); mainSheet.getRange("G8:H8").setValue(data.connName1); mainSheet.getRange("G9:H9").setValue(data.connName2); // G33:H33에 제출시간 기록 mainSheet.getRange("G33:H33").merge().setValue(submitTime); // --- [파트 C] 증명사진 및 서명 삽입 --- if (data.photo) { var photoBlob = convertBase64ToBlob(data.photo, data.nameKo + "_사진.png"); var photoRange = mainSheet.getRange("A12:B12"); photoRange.merge(); var img = mainSheet.insertImage(photoBlob, photoRange.getColumn(), photoRange.getRow()); setTopMarginAndCenterImage(img, photoRange, mainSheet); } var signs = [data.sign1, data.sign2, data.sign3]; var ranges = ["C12:D12", "E12:F12", "G12:H12"]; for (var i = 0; i < 3; i++) { if (signs[i]) { var signBlob = convertBase64ToBlob(signs[i], data.nameKo + "_서명" + (i+1) + ".png"); var targetRange = mainSheet.getRange(ranges[i]); targetRange.merge(); var img = mainSheet.insertImage(signBlob, targetRange.getColumn(), targetRange.getRow()); setTopMarginAndCenterImage(img, targetRange, mainSheet); if (i + 1 == data.selectedSign) { targetRange.setBackground("#e8f0fe"); } else { targetRange.setBackground(null); } } } // --- [파트 D] 안전화/유니폼 시트에 데이터 정확하게 누적 저장 --- subSheet.appendRow([ submitTime, // A: 제출시간 data.department, // B: 소속 data.nameKo, // C: 이름 data.birthDate || "미입력", // D: 생년월일 data.phone, // E: 휴대폰 data.safetyShoe || "미입력", // F: 안전화사이즈 data.uniformTopSize || "미입력", // G: 유니폼사이즈-상 data.uniformBottomSize || "미입력" // H: 유니폼사이즈-하 ]); // --- [파트 E] 오직 '시트1'만 완벽하게 고정하여 1페이지 PDF 저장 --- var pdfFileName = "인트라넷 사용자 등록요청서(품질)_" + data.nameKo + ".pdf"; var folders = DriveApp.getFoldersByName("인트라넷 사용자 등록 요청서"); var folder; if (folders.hasNext()) { folder = folders.next(); } else { folder = DriveApp.createFolder("인트라넷 사용자 등록 요청서"); } var url = ss.getUrl().replace(/edit$/, ''); var sheetId = mainSheet.getSheetId(); var exportUrl = url + 'export?' + 'exportFormat=pdf&format=pdf' + '&size=7' + // A4 '&portrait=true' + // 세로 방향 '&fitw=true' + // 너비 맞춤 '&gridlines=false' + // 눈금선 제거 '&printtitle=false' + // 제목 제거 '&fzr=false' + '&gid=' + sheetId + // 👈 출력할 시트를 '시트1'로 철저히 고정 '&r1=0&r2=34' + // 1행 ~ 35행 범위 '&c1=0&c2=8'; // A열 ~ I열 범위 var response = UrlFetchApp.fetch(exportUrl, { headers: { 'Authorization': 'Bearer ' + ScriptApp.getOAuthToken(), 'MuteHttpExceptions': true } }); var blob = response.getBlob().setName(pdfFileName); folder.createFile(blob); return ContentService.createTextOutput(JSON.stringify({"result": "success"})).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({"result": "error", "message": error.toString()})).setMimeType(ContentService.MimeType.JSON); } } function setTopMarginAndCenterImage(image, range, sheet) { var startCol = range.getColumn(); var startRow = range.getRow(); var numCols = range.getNumColumns(); var numRows = range.getNumRows(); var totalWidth = 0; for (var c = 0; c < numCols; c++) { totalWidth += sheet.getColumnWidth(startCol + c); } var totalHeight = 0; for (var r = 0; r < numRows; r++) { totalHeight += sheet.getRowHeight(startRow + r); } var maxAllowedWidth = totalWidth * 0.75; var maxAllowedHeight = totalHeight * 0.75; var imgWidth = image.getWidth(); var imgHeight = image.getHeight(); var widthRatio = maxAllowedWidth / imgWidth; var heightRatio = maxAllowedHeight / imgHeight; var ratio = Math.min(widthRatio, heightRatio, 1); var newWidth = imgWidth * ratio; var newHeight = imgHeight * ratio; image.setWidth(newWidth); image.setHeight(newHeight); var offsetX = Math.max(0, (totalWidth - newWidth) / 2); var offsetY = 6; image.setColumnOffset(offsetX); image.setRowOffset(offsetY); } function convertBase64ToBlob(base64Data, fileName) { var parts = base64Data.split(";base64,"); var contentType = parts[0].split(":")[1]; var raw = Utilities.base64Decode(parts[1]); return Utilities.newBlob(raw, contentType, fileName); }
